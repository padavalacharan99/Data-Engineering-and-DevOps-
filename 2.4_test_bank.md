31. **Check if a given number is prime**  
    - Implement using loops and conditions.
ANSWER : 

#!/bin/bash
is_prime() {
  num=$1
  if [ $num -le 1 ]; then
    echo "$num is not a prime number."
    return
  fi
  for (( i=2; i<num; i++ ))
  do
    if [ $((num % i)) -eq 0 ]; then
      echo "$num is not a prime number."
      return
    fi
  done
  echo "$num is a prime number."
}
echo "Enter a number:"

39. **Convert a decimal number to binary using a loop**  
    - Example: `10` → `1010`.
ANSWER : 

#!/bin/bash
read -p "Enter a decimal number: " decimal
binary=""
while [ $decimal -gt 0 ]; do
  remainder=$((decimal % 2))
  binary="$remainder$binary"
  decimal=$((decimal / 2))
done
if [ -z "$binary" ]; then
  binary="0"
fi
echo "Binary equivalent: $binary"
